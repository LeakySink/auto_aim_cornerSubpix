class MultiTargetTracker:
    def __init__(self):
        self.targets = []  # EKF列表
        self.priorities = []

    def evaluate_priority(self, target):
        score = 0
        score -= target.distance * 0.5         # 距离越近优先
        score += target.threat_level * 0.3     # 威胁度
        score += target.fire_opportunity * 0.2 # 射击时机
        return score

    def select_best_target(self):
        return max(self.targets, key=self.evaluate_priority)
